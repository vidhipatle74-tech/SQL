CREATE TABLE COURSE (
    CID      NUMBER PRIMARY KEY,
    CNAME    VARCHAR(30),
    FEES     NUMBER CHECK ( FEES > 0 ),
    DURATION NUMBER CHECK ( DURATION <= 12 )
);

CREATE TABLE STUDENT (
    SID   NUMBER PRIMARY KEY,
    SNAME VARCHAR(30),
    MNO   NUMBER CHECK ( LENGTH(MNO) = 10 ),
    DOB   DATE,
    PAN   CHAR(10) CHECK ( LENGTH(PAN) = 10 ),
    CID   NUMBER,
    CONSTRAINT CID_FK1 FOREIGN KEY ( CID )
        REFERENCES COURSE ( CID )
);

CREATE TABLE TRAINER1
 (
    TID NUMBER UNIQUE,
    TNAME VARCHAR(20) ,
    MNO NUMBER CHECK(LENGTH(MNO)=10),
    DOB DATE ,
    DOJ DATE ,
    SAL NUMBER CHECK(SAL>0)
    CONSTRAINT CID_FK2 FOREIGN KEY(CID) REFERENCES COURSE1(CID)
)

CREATE TABLE PAYMENT 
(
    PID NUMBER ,
    AMOUNT NUMBER CHECK(LENGTH(AMOUNT>=0)),
    STATUS VARCHAR(20),
    PAYMODE VARCHAR(20),
    PDATE DATE,
    PRODATE DATE,
    SID NUMBER,
    CONSTRAINTS SID_FK1 FOREIGN KEY(SID) REFERENCES STUDENT()
);

CRAETE TABLE ATTENDENCE
(
    DATE DATE,
    STATUS VARCHAR(20),
    BATCHCODE VARCHAR(20),
    SID NUMBER ,
    CONSTRAINTS SID_FK FOREIGN KEY(SID) REFERENCES STUDENT(SID)
);
